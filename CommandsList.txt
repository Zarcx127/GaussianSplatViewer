// Make Commands
//
// "make" defaults to "make reload"
//
// [mode] = debug/release
// by default [mode] = debug
//
// 	in release mode shaders are baked into executable
//

make reload mode=[mode]
    run .exe after rebuilding objs, shaders and executable

make clean_run mode=[mode]
	deletes all shaders and obj for [mode]
    run .exe after rebuilding objs, shaders and executable

make build mode=[mode]
	builds objs, shaders and executable

make run
	clears console and runs executable

make release
	cleans up all release resources 
	builds release objs, shaders and executable
	creates a zip that contains the release executable

make clean
    deletes all spv and obj files
    deletes the main executable
	deletes the release package

make clean_obj mode=[mode]
    deletes all obj files for that [mode]

make clean_shaders
    deletes all spv files

make delete_exe
	deletes the main executable

make create [option] [path_to_file]/[file_name]
	*DO NOT INCLUDE FILE EXTENSION
    creates in /(source/include)/[path_to_file] either:
		1. [file_name].hpp
		2. [file_name].cpp
		3. both a [file_name].hpp and [file_name].cpp

	the respective options are:
		1. header
		2. source
		3. pair

	*sub folder will be created if it does not exist	
	e.g. to create Example.hpp and Example.cpp 
		 in sub folder of  source/tmp/ and source/tmp/
		 type "make create pair tmp/Example" and 
		 both /include/tmp/Example.hpp and
		 /source/tmp/Example.cpp will be created

make delete [option] [path_to_file]/[file_name]
	*DO NOT INCLUDE FILE EXTENSION
    deletes from /(source/include)/[path_to_file] either:
		1. [file_name].hpp
		2. [file_name].cpp
		3. both a [file_name].hpp and [file_name].cpp

	the respective options are:
		1. header
		2. source
		3. pair
	
	*sub folder will be deleted if it is empty
	e.g. to delete Example.hpp and Example.cpp 
		 in sub folder of  source/tmp/ and source/tmp/
		 type "make delete pair tmp/Example" and 
		 both /include/tmp/Example.hpp and
		 /source/tmp/Example.cpp will be deleted

make create_shader [option] [path_to_file]/[file_name].[file_type]
	the respective options are:
		1. header
		2. source

	[option]=header
		creates [file_name].[file_type] in
		/shader/include/[path_to_file]

    [option]=source 
		creates [file_name].[file_type] in 
		/shader/[path_to_file]
		*[path_to_file] must not contain "include"

make delete_header [path_to_file]/[file_name].[file_type]
	deletes [file_name].[file_type] in 
	/shader/[path_to_file]
